<Program> -> <DeclarationList>
<DeclarationList> -> <Declaration> <DeclarationList> | ε
<Declaration> -> <FunctionDecl>
<VarDecl> -> <Type> IDENTIFIER <VarDeclRest> ";"
<VarDeclRest> -> "=" <Expression> | ε
<Type> -> "int" | "float" | "double"
<FunctionDecl> -> <Type> IDENTIFIER "(" <ParamList> ")" <CompoundStmt>
<ParamList> -> <Param> <ParamListRest> | ε
<ParamListRest> -> "," <Param> <ParamListRest> | ε
<Param> -> <Type> IDENTIFIER
<ClassDecl> -> "class" IDENTIFIER "{" <MemberList> "}" ";"
<MemberList> -> <Member> <MemberList> | ε
<Member> -> <VarDecl>
<CompoundStmt> -> "{" <StmtList> "}"
<StmtList> -> <Statement> <StmtList> | ε
<Statement> -> <IfStmt> | <WhileStmt> | <ReturnStmt> | <ExpressionStmt> | <CompoundStmt> | <VarDecl>
<IfStmt> -> "if" "(" <Expression> ")" <Statement> <ElsePart> | "if" "(" <Expression> ")" <Statement>
<ElsePart> -> "else" <Statement>
<WhileStmt> -> "while" "(" <Expression> ")" <Statement>
<ReturnStmt> -> "return" <Expression> ";"
<ExpressionStmt> -> <Expression> ";"
<Expression> -> <AssignmentExpr>
<AssignmentExpr> -> <LogicalOrExpr> <AssignOp> <AssignmentExpr> | <LogicalOrExpr>
<AssignOp> -> "=" | "+=" | "-=" | "*=" | "/="
<LogicalOrExpr> -> <LogicalAndExpr> <LogicalOrRest>
<LogicalOrRest> -> " | | " <LogicalAndExpr> <LogicalOrRest> | ε
<LogicalAndExpr> -> <EqualityExpr> <LogicalAndRest>
<LogicalAndRest> -> "&&" <EqualityExpr> <LogicalAndRest> | ε
<EqualityExpr> -> <RelationalExpr> <EqRest>
<EqRest> -> "==" <RelationalExpr> <EqRest> | ε
<RelationalExpr> -> <AddExpr> <RelRest>
<RelRest> -> "<" <AddExpr> <RelRest> | ε
<AddExpr> -> <Term> <AddRest>
<AddRest> -> "+" <Term> <AddRest> | ε
<Term> -> <Factor> <TermRest>
<TermRest> -> "*" <Factor> <TermRest> | ε
<Factor> -> <Primary> | "-" <Factor> | "!" <Factor>
<Primary> -> <Literal> | IDENTIFIER | <FunctionCall>
<FunctionCall> -> IDENTIFIER "(" <ArgList> ")"
<ArgList> -> <Expression> <ArgListRest> | ε
<ArgListRest> -> "," <Expression> <ArgListRest> | ε
<Literal> -> NUMBER | STRING
